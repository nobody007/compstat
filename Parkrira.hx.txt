#여기서 만든 함수에 맞게 직선 번호 순서 바꿔주기
x<-x[c(1:7,8,10,11,13,9,12,14:20)]




#Network function

h<-function(x){
#초기값 설정
a<-rep(0,4)
b<-rep(0,4)
c<-0

#A,B와 한번에 연결되어 있는 지점 표시
for(i in 1:4)
{a[i]<-ifelse(x[i]==1,1,0)
 b[i]<-ifelse(x[i+16]==1,1,0)}

for(i in 1:3)
{#아래쪽 점이 A에서 도달가능하지 않을 때, (아래쪽) 또는 (오른쪽 아래 왼쪽) 직선을 통해 도달 가능한 지점 표시
 if(a[i+1]==0) {if(a[i]*x[i+4]==1 | a[i]*x[i+7]*x[13+i]*x[i+8]==1) a[i+1]<-1}
 if(i==1 & a[2]==0) {if(a[1]*x[12]*x[9]==1) a[2]<-1}

 #위쪽 점이 A에서 도달가능하지 않을 때, (위쪽) 또는 (오른쪽 위 왼쪽) 직선을 통해 도달 가능한 지점 표시
 if(a[4-i]==0) {if(a[5-i]*x[8-i]==1 | a[5-i]*x[12-i]*x[17-i]*x[11-i]==1) a[4-i]<-1}
 if(i==1 & a[3]==0) {if(a[4]*x[13]*x[10]==1) a[3]<-1}

 #B도 마찬가지로 수행
 if(b[i+1]==0) {if(b[i]*x[i+13]==1 | b[i]*x[i+7]*x[i+4]*x[i+8]==1) b[i+1]<-1}
 if(i==1 & b[2]==0) {if(b[1]*x[8]*x[12]==1) b[2]<-1}

 if(b[4-i]==0) {if(b[5-i]*x[17-i]==1 | b[5-i]*x[12-i]*x[8-i]*x[11-i]==1) b[4-i]<-1}
 if(i==1 & b[3]==0) {if(b[4]*x[11]*x[13]==1) b[3]<-1}
}


#A,B에서 도달 가능한 지점과 그 사이 직선이 연결되어 있으면 CONNECT(c=1)
for(i in 1:4) {if(c==0) c<-ifelse(a[i]*b[i]*x[i+7]==1,1,0)}
if(c==0) {c<-ifelse(a[1]*b[2]*x[12]==1,1,0)}
if(c==0) {c<-ifelse(a[4]*b[3]*x[13]==1,1,0)}

return(c)
}

h(x)
